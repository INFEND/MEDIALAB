cmake_minimum_required(VERSION 3.18)
project(MediaLabAvifUltra LANGUAGES C CXX)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(AVIF_CODEC_AOM LOCAL CACHE STRING "" FORCE)
set(AVIF_LIBYUV OFF CACHE STRING "" FORCE)
set(AVIF_BUILD_APPS OFF CACHE BOOL "" FORCE)
set(AVIF_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(AVIF_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory("${AVIF_SOURCE}" avif EXCLUDE_FROM_ALL)
add_library(medialab_avif_ultra SHARED avif_ultra.cpp)
target_compile_features(medialab_avif_ultra PRIVATE cxx_std_17)
target_link_libraries(medialab_avif_ultra PRIVATE avif jnigraphics m)
target_link_options(medialab_avif_ultra PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--exclude-libs,ALL" "-Wl,--no-undefined")
