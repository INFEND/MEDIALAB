cmake_minimum_required(VERSION 3.18)
project(MediaLabPhotoExtra LANGUAGES C CXX)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
foreach(flag JPEGXL_ENABLE_TOOLS JPEGXL_ENABLE_DOXYGEN JPEGXL_ENABLE_MANPAGES JPEGXL_ENABLE_BENCHMARK JPEGXL_ENABLE_EXAMPLES JPEGXL_ENABLE_JNI JPEGXL_ENABLE_SJPEG JPEGXL_ENABLE_OPENEXR JPEGXL_ENABLE_TCMALLOC JPEGXL_ENABLE_FUZZERS JPEGXL_ENABLE_DEVTOOLS)
  set(${flag} OFF CACHE BOOL "" FORCE)
endforeach()
set(JPEGXL_ENABLE_SKCMS ON CACHE BOOL "" FORCE)
add_subdirectory("${JXL_SOURCE}" jxl EXCLUDE_FROM_ALL)
add_library(oxi STATIC IMPORTED)
set_target_properties(oxi PROPERTIES IMPORTED_LOCATION "${OXI_LIBRARY}")
add_library(medialab_photo_extra SHARED photo_extra.cpp)
target_compile_features(medialab_photo_extra PRIVATE cxx_std_17)
target_link_libraries(medialab_photo_extra PRIVATE jxl oxi jnigraphics log dl m)
target_link_options(medialab_photo_extra PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--exclude-libs,ALL")
