cmake_minimum_required(VERSION 3.18)
project(MediaLabPhotoUltra LANGUAGES C CXX)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(JPEGLI_ENABLE_TOOLS OFF CACHE BOOL "" FORCE)
set(JPEGLI_ENABLE_JPEGLI_LIBJPEG OFF CACHE BOOL "" FORCE)
set(JPEGLI_ENABLE_DOXYGEN OFF CACHE BOOL "" FORCE)
set(JPEGLI_ENABLE_MANPAGES OFF CACHE BOOL "" FORCE)
set(JPEGLI_ENABLE_BENCHMARK OFF CACHE BOOL "" FORCE)
set(JPEGLI_ENABLE_JNI OFF CACHE BOOL "" FORCE)
set(JPEGLI_ENABLE_SJPEG OFF CACHE BOOL "" FORCE)
set(JPEGLI_ENABLE_OPENEXR OFF CACHE BOOL "" FORCE)
set(JPEGLI_ENABLE_TCMALLOC OFF CACHE BOOL "" FORCE)
set(JPEGLI_ENABLE_SKCMS ON CACHE BOOL "" FORCE)
set(JPEGLI_ENABLE_FUZZERS OFF CACHE BOOL "" FORCE)
add_subdirectory("${JPEGLI_SOURCE}" jpegli EXCLUDE_FROM_ALL)
foreach(flag WEBP_BUILD_ANIM_UTILS WEBP_BUILD_CWEBP WEBP_BUILD_DWEBP WEBP_BUILD_GIF2WEBP WEBP_BUILD_IMG2WEBP WEBP_BUILD_VWEBP WEBP_BUILD_WEBPINFO WEBP_BUILD_WEBPMUX WEBP_BUILD_EXTRAS WEBP_BUILD_LIBWEBPMUX WEBP_BUILD_WEBP_JS WEBP_BUILD_FUZZTEST)
  set(${flag} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory("${WEBP_SOURCE}" webp EXCLUDE_FROM_ALL)
add_library(medialab_photo_ultra SHARED photo_ultra.cpp)
target_compile_features(medialab_photo_ultra PRIVATE cxx_std_17)
target_include_directories(medialab_photo_ultra PRIVATE "${JPEGLI_SOURCE}" "${WEBP_SOURCE}/src")
target_link_libraries(medialab_photo_ultra PRIVATE jpegli-static webp jnigraphics)
target_link_options(medialab_photo_ultra PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--exclude-libs,ALL")
